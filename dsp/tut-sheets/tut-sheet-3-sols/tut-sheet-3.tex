	\item Answer the following:
		\begin{enumerate}
			\item Analog signal $x_a(t) = 4 + 2 \cos(150 \pi t + \pi / 3) + 4 \sin (350 \pi t)$ is sampled at a rate of 200 samples/sec to obtain the DT signal $x(n)$. Determing $x(n)$ and DTFT $Y(e^{j \omega})$.

				\textbf{Solution}
				\begin{align*}
					x(n) &= x_a(nT) \\
					&= 4 + 2 \cos(150 \pi n/200 + \pi / 3) + 4 \sin (350 \pi n/200) \\
					&= 4 + 2 \cos(3 \pi n/4 + \pi / 3) + 4 \sin (7 \pi n/4)
				\end{align*}
				DTFT is given by:
				\begin{align*}
					Y(e^{j \omega}) &= \sum_{n=-\infty}^{\infty} x(n) e^{-j \omega n} \\
					&= 4 \sum_{n=-\infty}^{\infty} e^{-j \omega n} + 2 \sum_{n=-\infty}^{\infty} \cos(3 \pi n/4 + \pi / 3) e^{-j \omega n} + 4 \sum_{n=-\infty}^{\infty} \sin (7 \pi n/4) e^{-j \omega n} \\
					&= 4 \cdot 2 \pi \sum_{k=-\infty}^{\infty} \delta(\omega - 2 \pi k) + 2 \cdot \pi \left[ \frac{1}{2} e^{j \pi / 3} \sum_{k=-\infty}^{\infty} \delta(\omega - 3 \pi / 4 - 2 \pi k) + \frac{1}{2} e^{-j \pi / 3} \sum_{k=-\infty}^{\infty} \delta(\omega + 3 \pi / 4 - 2 \pi k) \right] \\
					&\quad + 4 \cdot \pi \left[ \frac{1}{2j} \sum_{k=-\infty}^{\infty} \delta(\omega - 7 \pi / 4 - 2 \pi k) - \frac{1}{2j} \sum_{k=-\infty}^{\infty} \delta(\omega + 7 \pi / 4 - 2 \pi k) \right] \\
					&= 8 \pi \sum_{k=-\infty}^{\infty} \delta(\omega - 2 \pi k) + \pi e^{j \pi / 3} \sum_{k=-\infty}^{\infty} \delta(\omega - 3 \pi / 4 - 2 \pi k) + \pi e^{-j \pi / 3} \sum_{k=-\infty}^{\infty} \delta(\omega + 3 \pi / 4 - 2 \pi k) \\
					&\quad + 2 \pi \left[ \frac{1}{j} \sum_{k=-\infty}^{\infty} \delta(\omega - 7 \pi / 4 - 2 \pi k) - \frac{1}{j} \sum_{k=-\infty}^{\infty} \delta(\omega + 7 \pi / 4 - 2 \pi k) \right]
				\end{align*}
		\end{enumerate}
		
\end{enumerate}
